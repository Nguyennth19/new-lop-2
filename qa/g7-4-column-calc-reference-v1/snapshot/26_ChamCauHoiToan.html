/* Bộ chấm bài thuần dữ liệu, có thể tái sử dụng bởi các bài Toán khác. Không thao tác DOM, không ghi tiến độ. */ function danhGiaCauToan(q,read){ if(!q||!q.valid)return false; const val=k=>String(read(k)||'').trim(); if(q.type==='vertical'){ const ans=String(q.answer).padStart(2,'0'); const carry=(Number(q.n1)%10+Number(q.n2)%10)>=10?'1':'0'; return val('ones')===ans[1]&&val('tens')===ans[0]&&val('carry')===carry; } if(q.type==='mental')return val('answer')!==''&&Number(val('answer'))===Number(q.answer); if(q.type==='missing_digit')return Array.isArray(q.puzzle)&&q.puzzle.every((row,ri)=>row.every((v,ci)=>!v||typeof v!=='object'||val('p'+ri+ci)===String(v.ans))); if(q.type==='word_problem'){ const a=Number(val('n1')),b=Number(val('n2')); return val('n1')!==''&&val('n2')!==''&& ((a===Number(q.n1)&&b===Number(q.n2))||(a===Number(q.n2)&&b===Number(q.n1)))&& val('op')===q.op&&val('sum')!==''&&Number(val('sum'))===Number(q.answer)&& val('answer')!==''&&Number(val('answer'))===Number(q.answer); } return false; }
